isim = girdi("Adın ne? ")
eğer isim == "Hasan":
    yazdır("Hoş geldin, dilin sahibi!")
değilse:
    yazdır("Merhaba, " + isim)

için i içinde aralık(1, 4):
    yazdır(i, "x", i, "=", i * i)
